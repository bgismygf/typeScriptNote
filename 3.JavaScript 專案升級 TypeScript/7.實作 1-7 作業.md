```ts
import axios from "axios";
import type { AxiosResponse } from "axios";

const request = axios.create({
  baseURL: "https://api.vercel.app/",
});

type ChildTag = {
  id: string;
  name: string;
};

type Tag = {
  id: string;
  name: string;
  child: ChildTag[];
};


type Product = {
  id: number;
  title: string;
  content: string;
  url: string;
  tags: string[];
};


type GetProductParams = {
  tag: string;
};


export const apiGetTagsList = (): Promise<AxiosResponse<Tag[]>> => request.get("/nav/tags");


export const apiGetTagsProduct = (params: GetProductParams): Promise<AxiosResponse<Product[]>> =>
  request.get("/nav/tags/product", {
    params,
  });
```
