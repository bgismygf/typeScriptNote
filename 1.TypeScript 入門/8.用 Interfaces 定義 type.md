# 用 Interfaces 定義 type

### 寫法

```js
interface IPerson {
  firstName: string;
  lastName: string;
}

const mike:IPerson = { firstName: "Cheng", lastName: "Mike" };
```

### 物件內的物件

```js
interface IJobInfo {
  job: "Frontend Engineer";
  baseInfo: {
    name: string;
    age: number;
    address: string;
  };
}

const jobInfo: IJobInfo = {
  job: "Frontend Engineer",
  baseInfo: {
    name: "Mike",
    age: 18,
    address: "Taipei",
  },
};
```

### 物件內的陣列

```js
interface IArea {
  name: string;
  zip: string;
}

interface ICity {
  name: string;
  area: IArea[];

  // 或是
  // area: Array<IArea>;
}

const taipei: ICity = {
  name: "新北市",
  area: [
    {
      name: "萬里區",
      zip: "207",
    },
    {
      name: "金山區",
      zip: "208",
    },
  ],
};
```

### 繼承其他 interface

```js
interface IWidth {
  width: string;
}

interface IHeight {
  height: string;
}

interface IBoxFontSize {
  "font-size": string;
  "font-weight": string;
}

// extends 後方都是繼承的 interface
interface IBoxStyle extends IWidth, IHeight, IBoxFontSize {
  color: string;
  border: string;
  padding: string;
  margin: string;
}

const box: IBoxStyle = {
  width: "100px",
  height: "100px",
  color: "red",
  border: "4px solid #000",
  padding: "10px",
  margin: "10px",
  "font-size": "16px",
  "font-weight": "bold",
};
```
