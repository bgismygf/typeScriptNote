# 物件與聯合 Record

## 重點
- type 寫法
- interface 寫法

### 寫法

```js
// 資料
const subscribersUserInfo = {
  mike: {
    age: 18,
    address: "桃園"
  },
  jacky: {
    age: 20,
    address: "台北"
  },
  andy: {
    age: 22,
    address: "台中"
  }
};
```

#### type 寫法

```js
type TPhones = "mike" | "jacky" | "andy";

type TUserInfo = {
  age: number;
  address: string;
}

type TRecordPhones = Record<TPhones, TUserInfo>;

const subscribersUserInfo: TRecordPhones = {
  mike: {
    age: 18,
    address: "桃園"
  },
  jacky: {
    age: 20,
    address: "台北"
  },
  andy: {
    age: 22,
    address: "台中"
  }
};
```

#### interface 寫法

```js
type TPhones = "mike" | "jacky" | "andy";

interface IUserInfo {
  age: number;
  address: string;
}

type TRecordPhones = Record<TPhones, IUserInfo>;

const subscribersUserInfo: TRecordPhones = {
  mike: {
    age: 18,
    address: "桃園"
  },
  jacky: {
    age: 20,
    address: "台北"
  },
  andy: {
    age: 22,
    address: "台中"
  }
};
```
